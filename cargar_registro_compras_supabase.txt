// cargar_registro_compras_supabase.mjs
// Requiere: Node.js y @oai/artifact-tool.
// Ejecutar: node cargar_registro_compras_supabase.mjs

import { FileBlob, SpreadsheetFile } from "@oai/artifact-tool";

const SUPABASE_URL = "https://TU-PROYECTO.supabase.co";
const SUPABASE_ANON_KEY = "TU_CLAVE_PUBLICA_DE_SUPABASE";
const EXCEL_PATH = "C:/ruta/Simulador_SUNAT_EmpresaDemo_4.xlsx";
const TABLE_NAME = "registro_compras";

const headers = {
  apikey: SUPABASE_ANON_KEY,
  Authorization: `Bearer ${SUPABASE_ANON_KEY}`,
};

function excelDate(value) {
  if (value == null || value === "") return null;
  if (typeof value === "string") return value;
  return new Date(Date.UTC(1899, 11, 30) + Math.round(value * 86400000))
    .toISOString()
    .slice(0, 10);
}

function nullIfBlank(value) {
  return value == null || value === "" || value === "-" ? null : value;
}

function toRecord(row) {
  const number = Number(row[0]);
  return {
    periodo: row[1],
    fecha_emision: excelDate(row[2]),
    fecha_vcto_pago: excelDate(row[3]),
    tipo_cp: row[4],
    comprobante: row[5],
    serie: row[6],
    numero: row[7],
    tipo_doc_identidad: nullIfBlank(row[8]),
    nro_doc_identidad: nullIfBlank(row[9]),
    proveedor: row[10],
    bi_gravada: row[11],
    dscto_bi: row[12],
    igv_ipm: row[13],
    dscto_igv_ipm: row[14],
    mto_exonerado: row[15],
    mto_inafecto: row[16],
    isc: row[17],
    icbper: row[18],
    otros_tributos: row[19],
    total_cp: row[20],
    moneda: row[21],
    tipo_cambio: row[22],
    doc_mod_fecha_emision: excelDate(row[23]),
    doc_mod_tipo_cp: nullIfBlank(row[24]),
    doc_mod_serie: nullIfBlank(row[25]),
    doc_mod_numero: nullIfBlank(row[26]),
    tipo_proveedor: nullIfBlank(row[27]),
    detalle_operacion: nullIfBlank(row[28]),
    descripcion: nullIfBlank(row[29]),
    forma_pago: nullIfBlank(row[30]),
    motivo_nota_credito: nullIfBlank(row[31]),
    sujeto_detrac_retenc: String(row[32]).trim().toLowerCase() === "si",
    tipo_carga: nullIfBlank(row[33]),
    pct_aplicado: row[34],
    monto_detrac_retenc: row[35],
    nro_constancia_deposito: nullIfBlank(row[36]),
    anticipos: row[37],
    otros_cargos: row[38],
    monto_redondeo: row[39],
    src_key: `C${String(number - 1).padStart(5, "0")}`,
  };
}

const source = await FileBlob.load(EXCEL_PATH);
const workbook = await SpreadsheetFile.importXlsx(source);
const sheet = workbook.worksheets.getItem("Registro de Compras");

// A2:AO435 contiene los 434 comprobantes; la fila 436 está vacía.
const records = sheet.getRange("A2:AO435").values.map(toRecord);

// Obtiene las claves existentes para evitar duplicados.
const existingResponse = await fetch(
  `${SUPABASE_URL}/rest/v1/${TABLE_NAME}?select=src_key&limit=10000`,
  { headers },
);
if (!existingResponse.ok) {
  throw new Error(await existingResponse.text());
}
const existingKeys = new Set((await existingResponse.json()).map((row) => row.src_key));
const pending = records.filter((record) => !existingKeys.has(record.src_key));

// Inserta en grupos de 50 registros.
for (let start = 0; start < pending.length; start += 50) {
  const batch = pending.slice(start, start + 50);
  const response = await fetch(`${SUPABASE_URL}/rest/v1/${TABLE_NAME}`, {
    method: "POST",
    headers: {
      ...headers,
      "Content-Type": "application/json",
      Prefer: "return=representation",
    },
    body: JSON.stringify(batch),
  });
  if (!response.ok) {
    throw new Error(`Falló el lote ${start / 50 + 1}: ${await response.text()}`);
  }
  console.log(`Lote ${start / 50 + 1}: ${batch.length} registros insertados.`);
}

// Verificación final.
const verificationResponse = await fetch(
  `${SUPABASE_URL}/rest/v1/${TABLE_NAME}?select=src_key&limit=10000`,
  { headers },
);
const keysAfterUpload = new Set((await verificationResponse.json()).map((row) => row.src_key));
const missing = records.filter((record) => !keysAfterUpload.has(record.src_key));

console.log({
  sourceRecords: records.length,
  inserted: pending.length,
  missingSourceRecords: missing.length,
});

if (missing.length > 0) {
  throw new Error(`Faltan ${missing.length} registros después de la carga.`);
}
